import { useState } from "react";
import { Sidebar } from "./components/Sidebar/Sidebar";
import { Toolbar } from "./components/Toolbar/Toolbar";
import { BrainWorkspace } from "./workspaces/Brain/BrainWorkspace";
import { SystemsWorkspace } from "./workspaces/Systems/SystemsWorkspace";
import { ActivityWorkspace } from "./workspaces/Activity/ActivityWorkspace";
import { LabWorkspace } from "./workspaces/Lab/LabWorkspace";
import type { SectionId } from "./config/navigation";
import type { SearchResult } from "./config/searchIndex";
import "./App.css";

export default function App() {
  const [active, setActive] = useState<SectionId>("brain");
  const [searchQuery, setSearchQuery] = useState("");
  const [focusedRegionId, setFocusedRegionId] = useState<string | null>(null);
  const [focusRegionSequence, setFocusRegionSequence] = useState(0);
  const [selectedSystemId, setSelectedSystemId] = useState("executive-control");
  const [selectedExperimentId, setSelectedExperimentId] = useState("reward-learning");
  const [selectedProtocolId, setSelectedProtocolId] = useState("resting-state");
  const [labMode, setLabMode] = useState<"models" | "protocols">("models");

  const focusBrainRegion = (regionId: string) => {
    setFocusedRegionId(regionId);
    setFocusRegionSequence((sequence) => sequence + 1);
    setActive("brain");
  };

  const selectSearchResult = (result: SearchResult) => {
    setSearchQuery("");
    setActive(result.section);

    if (result.section === "brain") {
      focusBrainRegion(result.targetId);
    } else if (result.section === "systems") {
      setSelectedSystemId(result.targetId);
    } else if (result.id.startsWith("lab-model:")) {
      setSelectedExperimentId(result.targetId);
      setLabMode("models");
    } else if (result.id.startsWith("lab-protocol:")) {
      setSelectedProtocolId(result.targetId);
      setLabMode("protocols");
    }
  };

  return (
    <div className="app">
      <Sidebar active={active} onSelect={setActive} />
      <div className="mainArea">
        <Toolbar
          active={active}
          searchQuery={searchQuery}
          onSearchQueryChange={setSearchQuery}
          onSelectSearchResult={selectSearchResult}
        />
        <main className="workspace">
          {active === "brain" && (
            <BrainWorkspace
              searchQuery={searchQuery}
              focusRegionId={focusedRegionId}
              focusRegionSequence={focusRegionSequence}
            />
          )}
          {active === "systems" && (
            <SystemsWorkspace
              searchQuery={searchQuery}
              selectedSystemId={selectedSystemId}
              onSelectSystem={setSelectedSystemId}
              onFocusRegion={focusBrainRegion}
            />
          )}
          {active === "activity" && (
            <ActivityWorkspace searchQuery={searchQuery} onFocusRegion={focusBrainRegion} />
          )}
          {active === "lab" && (
            <LabWorkspace
              searchQuery={searchQuery}
              mode={labMode}
              onModeChange={setLabMode}
              selectedExperimentId={selectedExperimentId}
              onSelectExperiment={setSelectedExperimentId}
              selectedProtocolId={selectedProtocolId}
              onSelectProtocol={setSelectedProtocolId}
            />
          )}
        </main>
      </div>
    </div>
  );
}

import { useEffect, useRef, useState } from "react";
import { SearchIcon } from "../icons";
import { NAV_SECTIONS, type SectionId } from "../../config/navigation";
import { searchNeuroLab, type SearchResult } from "../../config/searchIndex";
import "./Toolbar.css";

interface ToolbarProps {
  active: SectionId;
  searchQuery: string;
  onSearchQueryChange: (query: string) => void;
  onSelectSearchResult: (result: SearchResult) => void;
}

export function Toolbar({
  active,
  searchQuery,
  onSearchQueryChange,
  onSelectSearchResult,
}: ToolbarProps) {
  const searchInputRef = useRef<HTMLInputElement>(null);
  const [searchFocused, setSearchFocused] = useState(false);
  const results = searchNeuroLab(searchQuery);

  useEffect(() => {
    const handleShortcut = (event: KeyboardEvent) => {
      if ((event.metaKey || event.ctrlKey) && event.key.toLowerCase() === "k") {
        event.preventDefault();
        searchInputRef.current?.focus();
      }
    };

    window.addEventListener("keydown", handleShortcut);
    return () => window.removeEventListener("keydown", handleShortcut);
  }, []);

  const section = NAV_SECTIONS.find((item) => item.id === active);
  if (!section) {
    return null;
  }
  return (
    <header className="toolbar">
      <div className="context">
        <span className="contextSection">{section.label}</span>
        <span className="contextDivider">/</span>
        <span className="contextSub">{section.description}</span>
      </div>

      <div className="searchWrap">
        <SearchIcon size={15} className="searchIcon" />
        <input
          ref={searchInputRef}
          className="searchInput"
          type="text"
          placeholder="Search anatomy, systems, and concepts…"
          value={searchQuery}
          onChange={(event) => onSearchQueryChange(event.target.value)}
          onFocus={() => setSearchFocused(true)}
          onBlur={() => setSearchFocused(false)}
          onKeyDown={(event) => {
            if (event.key === "Escape") {
              setSearchFocused(false);
              searchInputRef.current?.blur();
            } else if (event.key === "Enter" && results[0]) {
              onSelectSearchResult(results[0]);
              setSearchFocused(false);
            }
          }}
          aria-label="Search all NeuroLab anatomy, systems, and concepts"
          aria-autocomplete="list"
          aria-expanded={searchFocused && searchQuery.trim().length > 0}
          aria-controls="globalSearchResults"
        />
        <span className="searchHint">Ctrl/⌘K</span>
        {searchFocused && searchQuery.trim() && (
          <div className="searchResults" id="globalSearchResults" role="listbox" aria-label="Global search results">
            {results.length > 0 ? results.map((result) => (
              <button
                type="button"
                key={result.id}
                className="searchResult"
                role="option"
                aria-selected="false"
                onMouseDown={(event) => event.preventDefault()}
                onClick={() => {
                  onSelectSearchResult(result);
                  setSearchFocused(false);
                }}
              >
                <span className="searchResultCopy">
                  <strong>{result.title}</strong>
                  <span>{result.description}</span>
                </span>
                <span className="searchResultCategory">{result.category}</span>
              </button>
            )) : (
              <p className="searchNoResults">No anatomy, systems, or concepts match.</p>
            )}
          </div>
        )}
      </div>
    </header>
  );
}

import type { SectionId } from "./navigation";
import { BRAIN_REGIONS, findBrainRegion, type BrainRegion } from "../workspaces/Brain/brainData";
import { NEURAL_SYSTEMS } from "../workspaces/Systems/systemsData";
import { EXPERIMENT_TEMPLATES, LEARNING_EXPERIMENTS } from "../workspaces/Lab/experimentData";

export interface SearchResult {
  id: string;
  title: string;
  category: string;
  description: string;
  section: SectionId;
  targetId: string;
  searchText: string;
}

function flattenRegions(regions: BrainRegion[]): BrainRegion[] {
  return regions.flatMap((region) => [region, ...flattenRegions(region.children ?? [])]);
}

const brainRegions = flattenRegions(BRAIN_REGIONS)
  .filter((region) => region.details)
  .map((region): SearchResult => ({
    id: `brain:${region.id}`,
    title: region.label,
    category: "Brain anatomy",
    description: region.details?.location ?? "Anatomical structure",
    section: "brain",
    targetId: region.id,
    searchText: [
      region.label,
      region.abbr,
      region.details?.location,
      region.details?.description,
      ...(region.details?.functions ?? []),
      ...(region.details?.systems ?? []),
    ].join(" ").toLowerCase(),
  }));

const systems = NEURAL_SYSTEMS.map((system): SearchResult => ({
  id: `systems:${system.id}`,
  title: system.name,
  category: "Neural system",
  description: system.overview,
  section: "systems",
  targetId: system.id,
  searchText: [
    system.name,
    system.shortName,
    system.overview,
    ...system.functions,
    ...system.regions,
    ...system.regions.map((id) => findBrainRegion(id)?.region.label ?? id),
    ...system.interactions,
  ].join(" ").toLowerCase(),
}));

const learningExperiments = LEARNING_EXPERIMENTS.map((experiment): SearchResult => ({
  id: `lab-model:${experiment.id}`,
  title: experiment.name,
  category: "Learning model",
  description: experiment.description,
  section: "lab",
  targetId: experiment.id,
  searchText: [experiment.name, experiment.description, ...experiment.keywords].join(" ").toLowerCase(),
}));

const protocols = EXPERIMENT_TEMPLATES.map((template): SearchResult => ({
  id: `lab-protocol:${template.id}`,
  title: template.name,
  category: "Protocol preview",
  description: template.description,
  section: "lab",
  targetId: template.id,
  searchText: [
    template.name,
    template.protocol,
    template.category,
    template.code,
    template.description,
  ].join(" ").toLowerCase(),
}));

const activity: SearchResult = {
  id: "activity:overview",
  title: "Illustrative brain activity",
  category: "Activity",
  description: "Synthetic activity indicators, network connections, and multichannel traces.",
  section: "activity",
  targetId: "activity",
  searchText: "activity activation signal synthetic waveform network connections intensity brain dynamics",
};

const SEARCH_INDEX = [...brainRegions, ...systems, ...learningExperiments, ...protocols, activity];

export function searchNeuroLab(query: string, limit = 7): SearchResult[] {
  const terms = query.trim().toLowerCase().split(/\s+/).filter(Boolean);
  if (terms.length === 0) return [];

  return SEARCH_INDEX
    .map((result, index) => {
      if (!terms.every((term) => result.searchText.includes(term))) {
        return { result, index, score: -1 };
      }
      const score = terms.reduce(
        (total, term) => total + (result.title.toLowerCase().startsWith(term) ? 3 : 1),
        0,
      );
      return { result, index, score };
    })
    .filter((item) => item.score >= 0)
    .sort((left, right) => right.score - left.score || left.index - right.index)
    .slice(0, limit)
    .map(({ result }) => result);
}

export function ActivityWorkspace({ searchQuery, onFocusRegion }: ActivityWorkspaceProps) {
  const [isRunning, setIsRunning] = useState(true);
  const [phase, setPhase] = useState(0);
  const [selectedChannel, setSelectedChannel] = useState("fp1");
  const [windowSeconds, setWindowSeconds] = useState(5);
  const [gain, setGain] = useState(1);
  const [activityIntensity, setActivityIntensity] = useState(62);
  const [selectedSystemId, setSelectedSystemId] = useState(NEURAL_SYSTEMS[0].id);
  const [selectedRegionId, setSelectedRegionId] = useState("prefrontal-cortex");

  const activeRegions = new Set(selectedSystem.regions);
  const selectedRegion = findBrainRegion(selectedRegionId)?.region;

  useEffect(() => {
    if (!isRunning) return;
    const timer = window.setInterval(() => setPhase((current) => current + 0.045), 50);
    return () => window.clearInterval(timer);
  }, [isRunning]);

  useEffect(() => {
    if (!selectedSystem.regions.includes(selectedRegionId)) {
      setSelectedRegionId(selectedSystem.regions[0]);
    }
  }, [selectedRegionId, selectedSystem]);

  const traces = useMemo(() => CHANNELS.map((channel, row) => {
    const centerY = 62 + row * 84;
    const points = Array.from({ length: POINT_COUNT }, (_, index) => {
      const progress = index / (POINT_COUNT - 1);
      const t = progress * windowSeconds * 2.4 + phase * channel.frequency;
      const signal =
        Math.sin(t * channel.frequency + channel.phase) * 0.48 +
        Math.sin(t * channel.frequency * 2.27 + channel.phase * 0.7) * 0.2 +
        Math.sin(t * 0.72 + channel.phase) * 0.22 +
        Math.sin(t * 5.1 + channel.phase * 1.3) * 0.07;
      const x = PLOT_LEFT + progress * (PLOT_RIGHT - PLOT_LEFT);
      const y = centerY - signal * 32 * gain * (0.65 + activityIntensity / 100);
      return `${x.toFixed(1)},${y.toFixed(1)}`;
    }).join(" ");
    return { ...channel, points, centerY, row };
  }), [phase, windowSeconds, gain, activityIntensity]);

export interface BrainRegionEntry {
  region: BrainRegion;
  parent: BrainRegion | null;
}

export function findBrainRegion(id: string): BrainRegionEntry | null {
  const findIn = (
    regions: BrainRegion[],
    parent: BrainRegion | null,
  ): BrainRegionEntry | null => {
    for (const region of regions) {
      if (region.id === id) return { region, parent };
      const match = findIn(region.children ?? [], region);
      if (match) return match;
    }
    return null;
  };

  return findIn(BRAIN_REGIONS, null);
}

export function getBrainStructure(id: string): BrainRegion | null {
  return findBrainRegion(id)?.region.details ? findBrainRegion(id)?.region ?? null : null;
}

import { Suspense, useEffect, useMemo, useState } from "react";
import { Canvas, type ThreeEvent } from "@react-three/fiber";
import { Html, OrbitControls, useGLTF } from "@react-three/drei";
import type { OrbitControls as OrbitControlsImpl } from "three-stdlib";
import { Color, Mesh, MeshStandardMaterial, type Object3D } from "three";
import type { RefObject } from "react";
import { findBrainRegion, type BrainRegion } from "./brainData";
import {
  BRAIN_MESH_REGIONS,
  BRAIN_MODEL_URL,
  type BrainMeshRegion,
} from "./brainVisualizationData";

interface BrainVisualizationProps {
  selectedRegionId: string | null;
  revealDeepStructures: boolean;
  onSelectRegion: (region: BrainRegion) => void;
  onHoverRegion: (label: string | null) => void;
  onZoomChange: (distance: number) => void;
  controlsRef: RefObject<OrbitControlsImpl>;
}

function RegionMeshes({
  region,
  object,
  selected,
  revealDeepStructures,
  onSelect,
  onHover,
}: {
  region: BrainMeshRegion;
  object: Object3D;
  selected: boolean;
  revealDeepStructures: boolean;
  onSelect: () => void;
  onHover: (label: string | null) => void;
}) {
  const [hovered, setHovered] = useState(false);

  useEffect(() => {
    object.traverse((child) => {
      if (!(child instanceof Mesh)) return;
      const materials = Array.isArray(child.material) ? child.material : [child.material];
      for (const material of materials) {
        if (!(material instanceof MeshStandardMaterial)) continue;
        const color = new Color(region.color);
        if (selected) color.lerp(new Color("#ffffff"), 0.32);
        else if (hovered) color.lerp(new Color("#ffffff"), 0.16);
        material.color.copy(color);
        material.emissive.set(region.color);
        material.emissiveIntensity = selected ? 0.42 : hovered ? 0.14 : 0.025;
        material.transparent = region.surface && revealDeepStructures;
        material.opacity = region.surface && revealDeepStructures ? (selected ? 0.58 : 0.16) : 1;
        material.depthWrite = !(region.surface && revealDeepStructures);
        material.needsUpdate = true;
      }
    });
  }, [hovered, object, region.color, region.surface, revealDeepStructures, selected]);

  return (
    <primitive
      object={object}
      onClick={(event: ThreeEvent<MouseEvent>) => {
        event.stopPropagation();
        onSelect();
      }}
      onPointerOver={(event: ThreeEvent<PointerEvent>) => {
        event.stopPropagation();
        setHovered(true);
        onHover(region.label);
        document.body.style.cursor = "pointer";
      }}
      onPointerOut={() => {
        setHovered(false);
        onHover(null);
        document.body.style.cursor = "auto";
      }}
    />
  );
}

function ModelLoadingLabel() {
  return (
    <Html center>
      <span className="brainLoadingLabel">LOADING SEGMENTED ATLAS MESH…</span>
    </Html>
  );
}

function BrainScene({
  selectedRegionId,
  revealDeepStructures,
  onSelectRegion,
  onHoverRegion,
  onZoomChange,
  controlsRef,
}: BrainVisualizationProps) {
  const { scene } = useGLTF(BRAIN_MODEL_URL);
  const atlasRoot = scene.getObjectByName("bodyparts3d-atlas");
  if (!atlasRoot) {
    throw new Error("The BodyParts3D atlas asset is missing its expected root node.");
  }

  const regionObjects = useMemo(
    () =>
      BRAIN_MESH_REGIONS.map((region) => {
        const source = atlasRoot.getObjectByName(region.id);
        if (!source) {
          throw new Error(`The BodyParts3D atlas asset is missing the ${region.label} segment.`);
        }
        const object = source.clone(true);
        object.traverse((child) => {
          if (child instanceof Mesh) {
            child.material = Array.isArray(child.material)
              ? child.material.map((material) => material.clone())
              : child.material.clone();
          }
        });
        return { region, object };
      }),
    [atlasRoot],
  );

  return (
    <>
      <fog attach="fog" args={["#090d14", 8, 15]} />
      <ambientLight intensity={0.62} />
      <hemisphereLight args={["#d8f1ff", "#111321", 0.95]} />
      <directionalLight
        position={[3.5, 5, 4]}
        color="#fff4e8"
        intensity={2.2}
        castShadow
        shadow-mapSize-width={1024}
        shadow-mapSize-height={1024}
      />
      <directionalLight position={[-4, 1, -3]} color="#6e89ff" intensity={1.2} />
      <pointLight position={[0, -2.5, 3]} color="#4fd9d4" intensity={16} distance={8} />

      <group position={atlasRoot.position} rotation={atlasRoot.rotation} scale={atlasRoot.scale}>
        {regionObjects.map(({ region, object }) => {
          const entry = findBrainRegion(region.id)?.region;
          if (!entry?.details) return null;
          return (
            <RegionMeshes
              key={region.id}
              region={region}
              object={object}
              selected={selectedRegionId === region.id}
              revealDeepStructures={revealDeepStructures}
              onSelect={() => onSelectRegion(entry)}
              onHover={onHoverRegion}
            />
          );
        })}
      </group>

      <OrbitControls
        ref={controlsRef}
        makeDefault
        enableRotate
        enableZoom
        enablePan
        enableDamping
        dampingFactor={0.075}
        minDistance={2.5}
        maxDistance={8}
        minPolarAngle={0.08}
        maxPolarAngle={Math.PI - 0.08}
        onChange={() => {
          const distance = controlsRef.current?.getDistance();
          if (distance) onZoomChange(distance);
        }}
      />
    </>
  );
}

    </Canvas>
  );
}

import { useEffect, useRef, useState } from "react";
import type { OrbitControls as OrbitControlsImpl } from "three-stdlib";
import { Panel, PanelSection } from "../../components/Panel/Panel";
import {
  InfoIcon,
  CrosshairIcon,
  ZoomInIcon,
  ZoomOutIcon,

import { BrainVisualization } from "./BrainVisualization";

  BRAIN_DEEP_REGION_IDS,
  BRAIN_MESH_REGION_BY_ID,
} from "./brainVisualizationData";
import "./BrainWorkspace.css";

interface BrainWorkspaceProps {
  searchQuery: string;
  focusRegionId?: string | null;
  focusRegionSequence?: number;
}

const MIN_ZOOM = 0.65;
const MAX_ZOOM = 1.8;

export function BrainWorkspace({
  searchQuery,
  focusRegionId,
  focusRegionSequence,
}: BrainWorkspaceProps) {
  const [selectedRegionId, setSelectedRegionId] = useState<string | null>(null);
  const [zoom, setZoom] = useState(1);
  const [showCrosshair, setShowCrosshair] = useState(false);
  const [showModel, setShowModel] = useState(true);
  const [revealDeepStructures, setRevealDeepStructures] = useState(false);
  const [hoveredRegionLabel, setHoveredRegionLabel] = useState<string | null>(null);
  const controlsRef = useRef<OrbitControlsImpl>(null);

  useEffect(() => {
    if (!focusRegionId) return;
    setSelectedRegionId(focusRegionId);
    if (BRAIN_DEEP_REGION_IDS.has(focusRegionId)) {
      setRevealDeepStructures(true);
    }
  }, [focusRegionId, focusRegionSequence]);

  const selectedEntry = selectedRegionId ? findBrainRegion(selectedRegionId) : null;
  const selectedRegion = selectedEntry?.region.details ? selectedEntry.region : null;
  const selectedMeshRegion = selectedRegionId
    ? BRAIN_MESH_REGION_BY_ID.get(selectedRegionId)
    : undefined;
  const details = selectedRegion?.details;

  const changeZoom = (amount: number) => {
    const controls = controlsRef.current;
    if (!controls) return;

    if (amount > 0) controls.dollyOut(1.2);
    else controls.dollyIn(1.2);
    setZoom(Number(Math.max(0.65, Math.min(1.8, 5.2 / controls.getDistance())).toFixed(2)));
  };

  const selectRegion = (region: BrainRegion) => {
    setSelectedRegionId(region.id);
    if (BRAIN_DEEP_REGION_IDS.has(region.id)) {
      setRevealDeepStructures(true);
    }
  };
  const setCameraView = (view: "lateral" | "anterior" | "superior") => {
    const controls = controlsRef.current;
    if (!controls) return;
    controls.object.up.set(0, 1, 0);
    if (view === "lateral") controls.object.position.set(5.2, 0.14, 0);
    else if (view === "anterior") controls.object.position.set(0, 0.14, 5.2);
    else {
      controls.object.position.set(0, 5.2, 0);
      controls.object.up.set(0, 0, -1);
    }
    controls.target.set(0, 0, 0);
    controls.update();
  };
  const resetView = () => {
    setZoom(1);
    controlsRef.current?.object.up.set(0, 1, 0);
    controlsRef.current?.reset();
  };

  return (
    <div className="brainWorkspace">
      <aside className="brainLeftPanel">
        <Panel title="Anatomy index" className="workspacePanel">
          <PanelSection title="Brain structures">
            <RegionTree
              selectedId={selectedRegionId}
              searchQuery={searchQuery}
              onSelect={selectRegion}
            />
            <p className="dataNotice">
              Drag to rotate · scroll to zoom · right-drag to pan. Lobe groups include selected gyri only.
            </p>
          </PanelSection>
          <div className="panelDivider" />
          <PanelSection title="View orientation">
            <div className="orientationCard">
              <span className="orientationAxis axis-front">A</span>
              <span className="orientationAxis axis-back">P</span>
              <span className="orientationAxis axis-left">L</span>
              <span className="orientationAxis axis-right">R</span>
              <span className="orientationCenter">●</span>
            </div>
            <div className="orientationCaption">ANTERIOR · POSTERIOR · L/R</div>
            <div className="brainViewPresets" role="group" aria-label="Preset viewing angles">

import { useState } from "react";
import { ChevronRightIcon } from "../../components/icons";
import { BRAIN_REGIONS, type BrainRegion } from "./brainData";
import { BRAIN_MESH_REGION_BY_ID } from "./brainVisualizationData";

interface RegionTreeProps {
  selectedId: string | null;
  searchQuery: string;
  onSelect: (region: BrainRegion) => void;
}

function filterRegions(regions: BrainRegion[], query: string): BrainRegion[] {
  return regions.reduce<BrainRegion[]>((matches, region) => {
    const matchingChildren = filterRegions(region.children ?? [], query);
    const matchesQuery = [region.label, region.abbr, region.id].some((value) =>
      value.toLowerCase().includes(query),
    );

    if (matchesQuery) {
      matches.push(region);
    } else if (matchingChildren.length > 0) {
      matches.push({ ...region, children: matchingChildren });
    }

    return matches;
  }, []);
}

export function RegionTree({ selectedId, searchQuery, onSelect }: RegionTreeProps) {
  const [expanded, setExpanded] = useState<Set<string>>(
    new Set(["cerebral-cortex", "subcortical-structures", "posterior-fossa"]),
  );
  const normalizedQuery = searchQuery.trim().toLowerCase();
  const regions = normalizedQuery
    ? filterRegions(BRAIN_REGIONS, normalizedQuery)
    : BRAIN_REGIONS;

  const toggle = (id: string) => {
    setExpanded((prev) => {
      const next = new Set(prev);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });
  };

  const renderRegion = (region: BrainRegion) => {
    const hasChildren = Boolean(region.children?.length);
    const isOpen = hasChildren && (
      normalizedQuery.length > 0 || expanded.has(region.id)
    );
    const isActive = selectedId === region.id;
    const meshRegion = BRAIN_MESH_REGION_BY_ID.get(region.id);

    return (
      <div key={region.id} className="regionGroup">
        <button
          type="button"
          className={`regionRow ${isActive ? "active" : ""}`}
          aria-pressed={isActive}
          aria-expanded={hasChildren ? isOpen : undefined}
          onClick={() => {
            if (hasChildren) toggle(region.id);
            if (region.details) onSelect(region);
          }}
        >
          <ChevronRightIcon
            size={12}
            className={`regionChevron ${isOpen ? "open" : ""}`}
            style={{ opacity: hasChildren ? 1 : 0 }}
          />
          <span
            className="regionDot"
            style={{ color: meshRegion?.color ?? region.color ?? "var(--text-muted)" }}
          />
          <span className="regionLabel">{region.label}</span>
          {region.details && !meshRegion && (
            <span className="regionMeshStatus" title="No independent mesh segment in this atlas">
              INFO
            </span>
          )}
          <span className="regionAbbr">{region.abbr}</span>
        </button>
        {hasChildren && isOpen && (
          <div className="regionChildren">
            {region.children?.map((child) => renderRegion(child))}
          </div>
        )}
      </div>
    );
  };

  if (regions.length === 0) {
    return <p className="treeEmpty">No matching regions.</p>;
  }

  return <div className="regionList">{regions.map((region) => renderRegion(region))}</div>;
}

  selectedExperimentId,
  onSelectExperiment,
}: ExperimentModelsProps) {
  const [expectedOutcome, setExpectedOutcome] = useState(55);
  const [actualOutcome, setActualOutcome] = useState(78);
  const [learningRate, setLearningRate] = useState(0.35);
  const [encoding, setEncoding] = useState(72);
  const [consolidation, setConsolidation] = useState(68);
  const [retrievalDelay, setRetrievalDelay] = useState(7);
  const [stress, setStress] = useState(50);

  const visibleExperiments = LEARNING_EXPERIMENTS.filter((experiment) =>
    [experiment.name, experiment.description, ...experiment.keywords].some((value) =>
      value.toLowerCase().includes(searchQuery.trim().toLowerCase()),
    ),
  );
  const selectedExperiment = LEARNING_EXPERIMENTS.find((item) => item.id === selectedExperimentId)
    ?? LEARNING_EXPERIMENTS[0];
  const predictionError = actualOutcome - expectedOutcome;
  const updatedExpectation = clamp(expectedOutcome + learningRate * predictionError, 0, 100);
  const memoryStrength = Math.round(
    100 * (encoding / 100) * (0.35 + 0.65 * consolidation / 100) * Math.exp(-retrievalDelay / 35),
  );
  const attention = Math.round(Math.max(0, 100 - ((stress - 50) / 50) ** 2 * 100));
  const workingMemory = Math.round(Math.max(0, 100 - ((stress - 42) / 55) ** 2 * 100));

  const memoryCurve = useMemo(() => Array.from({ length: 41 }, (_, index) => {
    const delay = index * 1.5;
    const strength = 100 * (encoding / 100) * (0.35 + 0.65 * consolidation / 100) * Math.exp(-delay / 35);
    const x = 56 + (index / 40) * 650;
    const y = 244 - (strength / 100) * 185;
    return `${index === 0 ? "M" : "L"}${x.toFixed(1)} ${y.toFixed(1)}`;
  }).join(" "), [encoding, consolidation]);

  const stressCurves = useMemo(() => {
    const points = Array.from({ length: 51 }, (_, index) => {
      const level = index * 2;
      const x = 56 + (level / 100) * 650;
      const attentionScore = Math.max(0, 100 - ((level - 50) / 50) ** 2 * 100);
      const memoryScore = Math.max(0, 100 - ((level - 42) / 55) ** 2 * 100);
      const attentionY = 244 - (attentionScore / 100) * 185;
      const memoryY = 244 - (memoryScore / 100) * 185;
      return { x, attentionY, memoryY };
    });
    return {
      attention: points.map((point, index) => `${index === 0 ? "M" : "L"}${point.x.toFixed(1)} ${point.attentionY.toFixed(1)}`).join(" "),
      memory: points.map((point, index) => `${index === 0 ? "M" : "L"}${point.x.toFixed(1)} ${point.memoryY.toFixed(1)}`).join(" "),
    };
  }, []);

  const stressMarkerX = 56 + (stress / 100) * 650;
  const attentionMarkerY = 244 - (attention / 100) * 185;
  const memoryMarkerY = 244 - (workingMemory / 100) * 185;

  return (
    <div className="experimentModels">
      <header className="modelsHeader">
        <div>
          <div className="labEyebrow"><span className="modelPulse" /> LEARNING MODELS <span className="labEyebrowDivider">/</span> EDUCATIONAL</div>
          <h2>Explore a model</h2>
          <p>Change one or more inputs and observe how a simplified mathematical model responds.</p>
        </div>
        <div className="modelStatus">3 INTERACTIVE CONCEPTS <span>·</span> NO BIOLOGICAL DATA</div>
      </header>

      <nav className="experimentPicker" aria-label="Select an educational model">
        {visibleExperiments.map((experiment, index) => (
          <button
            type="button"
            className={`experimentPickerCard ${selectedExperimentId === experiment.id ? "experimentSelected" : ""}`}
            key={experiment.id}
            onClick={() => onSelectExperiment(experiment.id)}
            aria-pressed={selectedExperimentId === experiment.id}
          >
            <span className="experimentPickerIndex">{String(index + 1).padStart(2, "0")}</span>
            <strong>{experiment.name}</strong>
            <span>{experiment.description}</span>
          </button>
        ))}
        {visibleExperiments.length === 0 && <p className="experimentNoResults">No learning models match this search.</p>}
      </nav>

      <section className="experimentModelPanel" aria-live="polite">
        <div className="modelPanelHeader">
          <div>
            <span className="modelCode">CONCEPT MODEL · {selectedExperiment.id.toUpperCase().replace(/-/g, "_")}</span>
            <h3>{selectedExperiment.name}</h3>
            <p>{selectedExperiment.description}</p>
          </div>
          <span className="modelTag">EDUCATIONAL · NOT VALIDATED</span>
        </div>

        {selectedExperiment.id === "reward-learning" && (
          <div className="modelContent">
            <div className="modelControls">
              <RangeControl label="EXPECTED OUTCOME" value={expectedOutcome} min={0} max={100} onChange={setExpectedOutcome} />
              <RangeControl label="ACTUAL OUTCOME" value={actualOutcome} min={0} max={100} onChange={setActualOutcome} />
              <RangeControl label="LEARNING RATE" value={learningRate} min={0} max={1} step={0.01} display={`${learningRate.toFixed(2)} α`} onChange={setLearningRate} />
              <div className={`predictionError ${predictionError >= 0 ? "positiveError" : "negativeError"}`}>
                <span>PREDICTION ERROR</span>
                <strong>{predictionError > 0 ? "+" : ""}{predictionError}</strong>
                <small>δ = actual − expected</small>
              </div>
            </div>
            <div className="modelVisualization">
              <div className="modelGraphTitle">SIMPLIFIED VALUE UPDATE <span>0–100 ARBITRARY UNITS</span></div>
              <svg className="rewardGraph" viewBox="0 0 760 270" role="img" aria-label={`Expected outcome ${expectedOutcome}, actual outcome ${actualOutcome}, updated expectation ${updatedExpectation.toFixed(0)}`}>
                <line x1="140" y1="46" x2="140" y2="226" className="modelGraphAxis" />
                <line x1="140" y1="226" x2="690" y2="226" className="modelGraphAxis" />
                {[0, 25, 50, 75, 100].map((value) => {
                  const x = 140 + value * 5.5;
                  return <g key={value}><line x1={x} y1="46" x2={x} y2="226" className="modelGraphGrid" /><text x={x} y="246" className="modelGraphTick">{value}</text></g>;
                })}
                <text x="15" y="92" className="modelBarLabel">EXPECTED</text>
                <text x="15" y="148" className="modelBarLabel">OBSERVED</text>
                <text x="15" y="204" className="modelBarLabel">UPDATED VALUE</text>
                <rect x="140" y="72" width={expectedOutcome * 5.5} height="27" rx="2" className="expectationBar" />
                <rect x="140" y="128" width={actualOutcome * 5.5} height="27" rx="2" className="outcomeBar" />
                <rect x="140" y="184" width={updatedExpectation * 5.5} height="27" rx="2" className="updatedBar" />
                <text x={Math.min(675, 147 + expectedOutcome * 5.5)} y="90" className="modelBarValue">{expectedOutcome}</text>
                <text x={Math.min(675, 147 + actualOutcome * 5.5)} y="146" className="modelBarValue">{actualOutcome}</text>
                <text x={Math.min(675, 147 + updatedExpectation * 5.5)} y="202" className="modelBarValue">{updatedExpectation.toFixed(0)}</text>
              </svg>
              <div className="modelEquation">updated expectation = expected + learning rate × prediction error</div>
            </div>
          </div>
        )}

        {selectedExperiment.id === "memory-formation" && (
          <div className="modelContent">
            <div className="modelControls">
              <RangeControl label="ENCODING STRENGTH" value={encoding} min={0} max={100} display={`${encoding}%`} onChange={setEncoding} />
              <RangeControl label="CONSOLIDATION OPPORTUNITY" value={consolidation} min={0} max={100} display={`${consolidation}%`} onChange={setConsolidation} />
              <RangeControl label="RETRIEVAL DELAY" value={retrievalDelay} min={0} max={60} display={`${retrievalDelay} days`} onChange={setRetrievalDelay} />
              <div className="memoryReadout">
                <span>CONCEPTUAL RETENTION</span>
                <strong>{memoryStrength}<small>%</small></strong>
                <div className="memoryStrengthTrack"><span style={{ width: `${memoryStrength}%` }} /></div>
              </div>
            </div>
            <div className="modelVisualization">
              <div className="modelGraphTitle">SIMPLIFIED RETENTION CURVE <span>CONCEPTUAL · ARBITRARY SCALE</span></div>
              <svg className="modelCurveGraph" viewBox="0 0 760 270" role="img" aria-label={`Conceptual memory retention curve with current strength ${memoryStrength} percent at ${retrievalDelay} days`}>
                <GraphGrid />
                <path d={memoryCurve} className="memoryCurve" />
                <line x1={56 + (retrievalDelay / 60) * 650} y1="55" x2={56 + (retrievalDelay / 60) * 650} y2="244" className="modelCurrentMarker" />
                <circle cx={56 + (retrievalDelay / 60) * 650} cy={244 - (memoryStrength / 100) * 185} r="5" className="memoryCurrentPoint" />
                <text x="56" y="260" className="modelGraphTick">0 DAYS</text>
                <text x="706" y="260" className="modelGraphTick">60 DAYS</text>
                <text x="65" y="48" className="modelGraphTick">STRONGER TRACE</text>
                <text x="65" y="238" className="modelGraphTick">WEAKER TRACE</text>
              </svg>
              <div className="memoryStages">
                <div><i className="encodingStage" /><span>ENCODING</span><b>{encoding}%</b></div>
                <div><i className="consolidationStage" /><span>CONSOLIDATION</span><b>{consolidation}%</b></div>
                <div><i className="retrievalStage" /><span>RETRIEVAL</span><b>{retrievalDelay} d</b></div>
              </div>
            </div>
          </div>
        )}

        {selectedExperiment.id === "stress-performance" && (
          <div className="modelContent">
            <div className="modelControls">
              <RangeControl label="CONCEPTUAL STRESS LEVEL" value={stress} min={0} max={100} display={`${stress}%`} onChange={setStress} />
              <div className="performanceReadout"><span>ATTENTION</span><strong>{attention}<small>/100</small></strong><div><i style={{ width: `${attention}%` }} /></div></div>
              <div className="performanceReadout workingMemoryReadout"><span>WORKING MEMORY</span><strong>{workingMemory}<small>/100</small></strong><div><i style={{ width: `${workingMemory}%` }} /></div></div>
              <p className="stressReadoutNote">Peak levels are set by this toy model and are not an individual prediction.</p>
            </div>
            <div className="modelVisualization">
              <div className="modelGraphTitle">ILLUSTRATIVE INVERTED-U RESPONSE <span>NOT A CLINICAL MEASURE</span></div>
              <svg className="modelCurveGraph" viewBox="0 0 760 270" role="img" aria-label={`Stress level ${stress}; modeled attention ${attention} and working memory ${workingMemory} out of 100`}>
                <GraphGrid />
                <path d={stressCurves.attention} className="attentionCurve" />
                <path d={stressCurves.memory} className="workingMemoryCurve" />
                <line x1={stressMarkerX} y1="55" x2={stressMarkerX} y2="244" className="modelCurrentMarker" />
                <circle cx={stressMarkerX} cy={attentionMarkerY} r="5" className="attentionCurrentPoint" />
                <circle cx={stressMarkerX} cy={memoryMarkerY} r="5" className="memoryCurrentPoint" />
                <text x="56" y="260" className="modelGraphTick">LOW STRESS</text>
                <text x="706" y="260" className="modelGraphTick">HIGH STRESS</text>
                <text x="65" y="48" className="modelGraphTick">PERFORMANCE</text>
              </svg>
              <div className="modelLegend"><span><i className="attentionLegend" />ATTENTION</span><span><i className="memoryLegend" />WORKING MEMORY</span></div>
            </div>
          </div>
        )}

        <div className="modelDisclaimer">
          <span>i</span>
          <p>These are educational mathematical demonstrations with deliberately simplified equations. They are not validated biological simulations, measurements, clinical guidance, or individual predictions.</p>
        </div>
      </section>
    </div>
  );
}

function RangeControl({
  label,
  value,
  min,
  max,
  step = 1,
  display = `${value}`,
  onChange,
}: {
  label: string;
  value: number;
  min: number;
  max: number;
  step?: number;
  display?: string;
  onChange: (value: number) => void;
}) {
  return (
    <label className="modelRangeControl">
      <span className="modelControlLabel">{label}<b>{display}</b></span>
      <input type="range" min={min} max={max} step={step} value={value} onChange={(event) => onChange(Number(event.target.value))} />
      <span className="modelRangeEnds"><small>{min}</small><small>{max}</small></span>
    </label>
  );
}

function GraphGrid() {
  return (
    <>
      {[0, 25, 50, 75, 100].map((value) => {
        const x = 56 + value * 6.5;
        const y = 244 - value * 1.85;
        return (
          <g key={value}>
            <line x1={x} y1="55" x2={x} y2="244" className="modelGraphGrid" />
            <line x1="56" y1={y} x2="706" y2={y} className="modelGraphGrid" />
          </g>
        );
      })}
      <line x1="56" y1="244" x2="706" y2="244" className="modelGraphAxis" />
      <line x1="56" y1="55" x2="56" y2="244" className="modelGraphAxis" />
    </>
  );
}

import { useEffect, useState } from "react";
import type { CSSProperties } from "react";
import { DownloadIcon, LabIcon, PauseIcon, PlayIcon, RotateIcon } from "../../components/icons";
import { ExperimentModels } from "./ExperimentModels";
import { EXPERIMENT_TEMPLATES } from "./experimentData";
import "./LabWorkspace.css";

const DURATIONS = [30, 60, 90];
const formatTime = (seconds: number) => `${Math.floor(seconds / 60).toString().padStart(2, "0")}:${Math.floor(seconds % 60).toString().padStart(2, "0")}`;

interface LabWorkspaceProps {
  searchQuery: string;
  mode: "models" | "protocols";
  onModeChange: (mode: "models" | "protocols") => void;
  selectedExperimentId: string;
  onSelectExperiment: (id: string) => void;
  selectedProtocolId: string;
  onSelectProtocol: (id: string) => void;
}

export function LabWorkspace({
  searchQuery,
  mode,
  onModeChange,
  selectedExperimentId,
  onSelectExperiment,
  selectedProtocolId,
  onSelectProtocol,
}: LabWorkspaceProps) {
  const [duration, setDuration] = useState(60);
  const [elapsed, setElapsed] = useState(0);
  const [isRunning, setIsRunning] = useState(false);
  const selectedTemplate = EXPERIMENT_TEMPLATES.find((template) => template.id === selectedProtocolId) ?? EXPERIMENT_TEMPLATES[0];
  const visibleTemplates = EXPERIMENT_TEMPLATES.filter((template) =>
    [template.name, template.protocol, template.category, template.code, template.description]
      .some((value) => value.toLowerCase().includes(searchQuery.trim().toLowerCase())),
  );
  const progress = Math.min(1, elapsed / duration);
  const activePhaseIndex = selectedTemplate.phases.findIndex((phase) => progress < phase.duration + selectedTemplate.phases.slice(0, selectedTemplate.phases.indexOf(phase)).reduce((sum, item) => sum + item.duration, 0));
  const activePhase = activePhaseIndex >= 0 ? selectedTemplate.phases[activePhaseIndex] : null;

  useEffect(() => {
    if (!isRunning || mode !== "protocols") return;
    const timer = window.setInterval(() => {
      setElapsed((current) => Math.min(duration, current + 1));
    }, 1000);
    return () => window.clearInterval(timer);
  }, [duration, isRunning, mode]);

  useEffect(() => {
    if (isRunning && mode === "protocols" && elapsed >= duration) {
      setIsRunning(false);
    }
  }, [duration, elapsed, isRunning, mode]);

  const chooseTemplate = (id: string) => {
    onSelectProtocol(id);
    setElapsed(0);
    setIsRunning(false);
  };

  const reset = () => {
    setElapsed(0);
    setIsRunning(false);
  };

  return (
    <div className="labWorkspace">
      <header className="labHeader">
        <div>
          <div className="labEyebrow"><span className="labEyebrowMark"><LabIcon size={13} /></span> EXPERIMENT DESIGN <span className="labEyebrowDivider">/</span> LOCAL PROTOCOLS</div>
          <h1>Experimental lab</h1>
          <p>Explore educational learning models or preview a local protocol timeline.</p>
        </div>
        <div className="localOnlyBadge"><span /> LOCAL PREVIEW ONLY</div>
      </header>

      <nav className="labModeTabs" aria-label="Lab workspace">
        <button type="button" className={mode === "models" ? "labModeActive" : ""} onClick={() => onModeChange("models")} aria-pressed={mode === "models"}>
          Learning models <span>03</span>
        </button>
        <button type="button" className={mode === "protocols" ? "labModeActive" : ""} onClick={() => onModeChange("protocols")} aria-pressed={mode === "protocols"}>
          Protocol preview <span>LOCAL</span>
        </button>
      </nav>

      {mode === "models" ? (
        <ExperimentModels
          searchQuery={searchQuery}
          selectedExperimentId={selectedExperimentId}
          onSelectExperiment={onSelectExperiment}
        />
      ) : (
      <div className="labLayout">
        <aside className="templatePanel">
          <div className="labSectionTitle">PROTOCOL LIBRARY <span>{String(EXPERIMENT_TEMPLATES.length).padStart(2, "0")}</span></div>
          <div className="templateList">
            {visibleTemplates.map((template, index) => (
              <button type="button" className={`templateCard ${selectedProtocolId === template.id ? "templateSelected" : ""}`} key={template.id} onClick={() => chooseTemplate(template.id)} aria-pressed={selectedProtocolId === template.id}>
                <span className="templateIndex">{String(index + 1).padStart(2, "0")}</span>
                <span className="templateCopy"><span className="templateCategory">{template.category} · {template.code}</span><strong>{template.name}</strong><span className="templateDescription">{template.description}</span></span>
                <span className="templateArrow">↗</span>
              </button>
            ))}
            {visibleTemplates.length === 0 && <p className="templateNoResults">No protocols match this search.</p>}
          </div>
          <div className="templateFootnote">Templates are illustrative examples, not validated experimental protocols.</div>
        </aside>

        <main className="protocolWorkspace">
          <div className="protocolHeading">
            <div className="protocolHeadingText"><div className="protocolCode">{selectedTemplate.code} <span>·</span> {selectedTemplate.category}</div><h2>{selectedTemplate.protocol}</h2><p>{selectedTemplate.description}</p></div>
            <button type="button" className="exportButton" disabled title="Export is unavailable in this prototype"><DownloadIcon size={14} /> EXPORT <span>SOON</span></button>
          </div>

          <section className="timelinePanel">
            <div className="timelineHeader"><span>PROTOCOL TIMELINE</span><span>{selectedTemplate.phases.length} PHASES <i /> {duration} S TOTAL</span></div>
            <div className="timelineTrack" aria-label="Protocol phases">
              {selectedTemplate.phases.map((phase, index) => {
                const start = selectedTemplate.phases.slice(0, index).reduce((sum, item) => sum + item.duration, 0);
                const phaseProgress = Math.max(0, Math.min(1, (progress - start) / phase.duration));
                const phaseState = progress >= start + phase.duration ? "complete" : progress >= start ? "phaseCurrent" : "";
                return (
                  <div className={`timelinePhase ${phaseState}`} key={phase.label} style={{ flex: phase.duration, "--phase-color": phase.color } as CSSProperties}>
                    <div className="phaseBar"><span style={{ transform: `scaleX(${phaseProgress})` }} /></div>
                    <div className="phaseIndex">{String(index + 1).padStart(2, "0")}</div>
                    <div className="phaseName">{phase.label}</div>
                    <div className="phaseDescription">{phase.description}</div>
                    {phaseState === "phaseCurrent" && <div className="phaseNow">NOW</div>}
                  </div>
                );
              })}
            </div>
            <div className="timelineProgress"><span style={{ width: `${progress * 100}%` }} /></div>
          </section>

          <section className="runPanel">
            <div className="runStatus">
              <span className={`runPulse ${isRunning ? "running" : ""}`} />
              <div><span className="runStatusLabel">SESSION STATE</span><strong>{isRunning ? "Preview running" : progress >= 1 ? "Preview complete" : "Ready to preview"}</strong></div>
              <span className="runTimer">{formatTime(elapsed)} <i>/</i> {formatTime(duration)}</span>
            </div>
            <div className="runProgressTrack"><span style={{ width: `${progress * 100}%` }} /></div>
            <div className="runActions">
              <span className="runContext">{activePhase ? `CURRENT PHASE  /  ${activePhase.label.toUpperCase()}` : progress >= 1 ? "ALL PHASES COMPLETE" : "PROTOCOL CONFIGURED"}</span>
              <div>
                <button type="button" className="runReset" onClick={reset} disabled={elapsed === 0 && !isRunning} aria-label="Reset preview"><RotateIcon size={15} /></button>
                <button type="button" className={`runButton ${isRunning ? "pauseButton" : ""}`} onClick={() => {
                  if (progress >= 1) setElapsed(0);
                  setIsRunning((running) => !running);
                }}>
                  {isRunning ? <PauseIcon size={14} /> : <PlayIcon size={14} />}
                  {isRunning ? "PAUSE PREVIEW" : progress >= 1 ? "RESTART PREVIEW" : "RUN PREVIEW"}
                </button>
              </div>
            </div>
          </section>

          <section className="labParameters">
            <div className="labSectionTitle">SESSION PARAMETERS <span>CONFIGURATION</span></div>
            <div className="parameterGrid">
              <div className="parameterField"><label htmlFor="preview-duration">PREVIEW DURATION</label><select id="preview-duration" value={duration} onChange={(event) => { setDuration(Number(event.target.value)); setElapsed(0); setIsRunning(false); }}>{DURATIONS.map((seconds) => <option key={seconds} value={seconds}>{seconds} seconds</option>)}</select></div>
              <div className="parameterField"><span>STIMULUS DELIVERY</span><div className="parameterValue">VISUAL TIMELINE ONLY <i>NO OUTPUT</i></div></div>
              <div className="parameterField"><span>DATA COLLECTION</span><div className="parameterValue">DISABLED <i>LOCAL ONLY</i></div></div>
            </div>
          </section>

          <div className="labDisclaimer"><span>i</span> This preview only advances a local timeline. No stimuli are presented, participant information is collected, or research data is recorded.</div>
        </main>
      </div>
      )}
    </div>
  );
}

import { useEffect, useMemo, useState } from "react";
import type { CSSProperties } from "react";
import { NEURAL_SYSTEMS, NETWORK_EDGES, NETWORK_NODES } from "./systemsData";
import { findBrainRegion } from "../Brain/brainData";
import "./SystemsWorkspace.css";

interface SystemsWorkspaceProps {
  searchQuery: string;
  selectedSystemId: string;
  onSelectSystem: (systemId: string) => void;
  onFocusRegion: (regionId: string) => void;
}

export function SystemsWorkspace({
  searchQuery,
  selectedSystemId,
  onSelectSystem,
  onFocusRegion,
}: SystemsWorkspaceProps) {
  const [selectedRegionId, setSelectedRegionId] = useState<string | null>(null);
  const [selectedConnection, setSelectedConnection] = useState<[string, string] | null>(null);
  const selectedSystem = NEURAL_SYSTEMS.find((system) => system.id === selectedSystemId) ?? NEURAL_SYSTEMS[0];

  useEffect(() => {
    setSelectedRegionId(null);
    setSelectedConnection(null);
  }, [selectedSystemId]);

  const visibleSystems = useMemo(() => {
    const query = searchQuery.trim().toLowerCase();
    return query
      ? NEURAL_SYSTEMS.filter((system) =>
          [
            system.name,
            system.overview,
            ...system.functions,
            ...system.regions.map((id) => findBrainRegion(id)?.region.label ?? id),
            ...system.interactions.map((id) => NEURAL_SYSTEMS.find((item) => item.id === id)?.name ?? id),
          ].some((value) => value.toLowerCase().includes(query)),
        )
      : NEURAL_SYSTEMS;
  }, [searchQuery]);
  const nodeById = new Map(NETWORK_NODES.map((node) => [node.id, node]));
  const activeRegions = new Set(selectedSystem.regions);
  const selectedRegion = selectedRegionId ? findBrainRegion(selectedRegionId)?.region : null;
  const selectedRegionDetails = selectedRegion?.details;
  const selectedLink = selectedConnection
    ? NETWORK_EDGES.find(([from, to]) => from === selectedConnection[0] && to === selectedConnection[1])
    : null;
  const selectedLinkRegions = selectedLink
    ? selectedLink.flatMap((id) => {
        const region = findBrainRegion(id)?.region;
        return region?.details ? [region] : [];
      })
    : [];
  const sharedSystems = selectedLinkRegions[0]?.details?.systems.filter((system) =>
    selectedLinkRegions[1]?.details?.systems.includes(system),
  ) ?? [];

  const selectRegion = (regionId: string) => {
    setSelectedRegionId(regionId);
    setSelectedConnection(null);
  };

  return (
    <div className="systemsWorkspace">
      <header className="systemsHeader">
        <div>
          <div className="eyebrow"><span className="liveIndicator" /> NETWORK ATLAS <span className="eyebrowDivider">/</span> 08 SYSTEMS</div>
          <h1>Neural systems</h1>
          <p>Explore interacting circuits across the brain, from regions to distributed function.</p>
        </div>
        <div className="systemsHeaderMeta"><span className="metaDot" /> INTERACTIVE CONNECTOME <span className="metaSeparator">·</span> INTRODUCTORY MODEL</div>
      </header>

      <div className="systemsLayout">
        <aside className="systemIndex">
          <div className="sectionLabel">FUNCTIONAL SYSTEMS <span>08</span></div>
          <nav aria-label="Neural systems">
            {visibleSystems.map((system, index) => (
              <button
                type="button"
                key={system.id}
                className={`systemIndexItem ${selectedSystemId === system.id ? "selected" : ""}`}
                style={{ "--system-color": system.color } as CSSProperties}
                onClick={() => onSelectSystem(system.id)}
                aria-pressed={selectedSystemId === system.id}
              >
                <span className="systemIndexNumber">{String(index + 1).padStart(2, "0")}</span>
                <span className="systemIndexName">{system.name}</span>
                <span className="systemIndexMark" />
              </button>
            ))}
            {visibleSystems.length === 0 && <p className="systemsNoResults">No systems match this search.</p>}
          </nav>
          <div className="indexFootnote">System boundaries are approximate; networks overlap and reconfigure with task and state.</div>
        </aside>

        <main className="systemDetail">
          <div className="detailHeading">
            <div className="systemNumber" style={{ color: selectedSystem.color }}>{String(NEURAL_SYSTEMS.findIndex((item) => item.id === selectedSystemId) + 1).padStart(2, "0")}</div>
            <div>
              <span className="detailEyebrow" style={{ color: selectedSystem.color }}>FUNCTIONAL SYSTEM</span>
              <h2>{selectedSystem.name}</h2>
            </div>
            <span className="regionCount">{selectedSystem.regions.length} NODES IN VIEW</span>
          </div>

          <section className="networkPanel" aria-label={`${selectedSystem.name} network visualization`}>
            <div className="networkPanelTop">
              <span>REGIONAL INTERACTIONS</span>
              <span className="networkHint">SELECT A NODE TO OPEN ANATOMY <span>↗</span></span>
            </div>
            <svg className="networkDiagram" viewBox="0 0 760 430" role="img" aria-label="Interactive diagram of brain region relationships">
              <defs>
                <radialGradient id="networkGlow">
                  <stop offset="0%" stopColor={selectedSystem.color} stopOpacity=".17" />
                  <stop offset="100%" stopColor={selectedSystem.color} stopOpacity="0" />
                </radialGradient>
              </defs>
              <ellipse cx="378" cy="218" rx="300" ry="190" fill="url(#networkGlow)" />
              <path className="networkOutline" d="M75 199 C78 119 133 63 231 63 C283 30 380 47 438 70 C523 57 642 90 675 161 C710 235 650 309 609 341 C546 399 430 405 366 374 C283 408 188 375 146 341 C91 311 60 255 75 199Z" />
              {NETWORK_EDGES.map(([from, to]) => {
                const start = nodeById.get(from);
                const end = nodeById.get(to);
                if (!start || !end) return null;
                const isActive = activeRegions.has(from) && activeRegions.has(to);
                const isSelected = selectedConnection?.[0] === from && selectedConnection[1] === to;
                return (
                  <g
                    key={`${from}-${to}`}
                    className="networkConnection"
                    role="button"
                    tabIndex={0}
                    aria-label={`Inspect relationship between ${start.label} and ${end.label}`}
                    aria-pressed={isSelected}
                    onClick={() => {
                      setSelectedConnection([from, to]);
                      setSelectedRegionId(null);
                    }}
                    onKeyDown={(event) => {
                      if (event.key === "Enter" || event.key === " ") {
                        event.preventDefault();
                        setSelectedConnection([from, to]);
                        setSelectedRegionId(null);
                      }
                    }}
                  >
                    <line className="networkEdgeHit" x1={start.x} y1={start.y} x2={end.x} y2={end.y} />
                    <line
                      x1={start.x}
                      y1={start.y}
                      x2={end.x}
                      y2={end.y}
                      className={`networkEdge ${isActive ? "edgeActive" : ""} ${isSelected ? "edgeSelected" : ""}`}
                      style={{ "--system-color": selectedSystem.color } as CSSProperties}
                    />
                  </g>
                );
              })}
              {NETWORK_NODES.map((node) => {
                const isActive = activeRegions.has(node.id);
                const isSelected = selectedRegionId === node.id;
                return (
                  <g
                    key={node.id}
                    className={`networkNode ${isActive ? "nodeActive" : ""} ${isSelected ? "nodeSelected" : ""}`}
                    transform={`translate(${node.x} ${node.y})`}
                    style={{ "--system-color": selectedSystem.color } as CSSProperties}
                    role="button"
                    tabIndex={0}
                    aria-label={`Inspect ${node.label}`}
                    aria-pressed={isSelected}
                    onClick={() => selectRegion(node.id)}
                    onKeyDown={(event) => {
                      if (event.key === "Enter" || event.key === " ") {
                        event.preventDefault();
                        selectRegion(node.id);
                      }
                    }}
                  >
                    <circle className="nodeHalo" r={isActive || isSelected ? 27 : 20} />
                    <circle className="nodeCircle" r={isActive || isSelected ? 7 : 5} />
                    <text className="nodeLabel" y={isActive ? 25 : 21}>{node.label}</text>
                  </g>
                );
              })}
              <text className="networkOrientation" x="30" y="24">ANTERIOR</text>
              <text className="networkOrientation" x="665" y="24">POSTERIOR</text>
            </svg>
            <div className="networkLegend"><span className="legendNode" /> SYSTEM-ASSOCIATED REGION <span className="legendLine" /> SELECT A LINK OR REGION TO INSPECT IT</div>
          </section>

          <section className="networkSelectionPanel" aria-live="polite">
            {selectedRegion && selectedRegionDetails ? (
              <>
                <div className="networkSelectionCopy">
                  <span className="detailEyebrow">BRAIN REGION · {selectedRegion.abbr}</span>
                  <h3>{selectedRegion.label}</h3>
                  <p>{selectedRegionDetails.description}</p>
                  <span className="selectionMeta">{selectedRegionDetails.location}</span>
                </div>
                <div className="networkSelectionActions">
                  <div className="selectionConnections">
                    {selectedRegionDetails.connections.map((id) => {
                      const connection = findBrainRegion(id)?.region;
                      return connection ? (
                        <button type="button" className="connectionMiniChip" key={id} onClick={() => selectRegion(id)}>
                          {connection.label}
                        </button>
                      ) : null;
                    })}
                  </div>
                  <button type="button" className="openBrainButton" onClick={() => onFocusRegion(selectedRegion.id)}>
                    Open in Brain <span>↗</span>
                  </button>
                </div>
              </>
            ) : selectedLink ? (
              <>
                <div className="networkSelectionCopy">
                  <span className="detailEyebrow">MODELED RELATIONSHIP</span>
                  <h3>{selectedLinkRegions[0]?.label} <span>↔</span> {selectedLinkRegions[1]?.label}</h3>
                  <div className="linkRegionPair">
                    {selectedLinkRegions.map((region) => (
                      <p key={region.id}><b>{region.label}:</b> {region.details?.description}</p>
                    ))}
                  </div>
                  {sharedSystems.length > 0 && <span className="selectionMeta">Shared systems: {sharedSystems.join(" · ")}</span>}
                  <span className="selectionMeta">This line is a conceptual relationship, not a measured pathway or claim of direct structural connectivity.</span>
                </div>
                <div className="networkSelectionActions">
                  {selectedLinkRegions.map((region) => (
                    <button type="button" className="openBrainButton" key={region.id} onClick={() => onFocusRegion(region.id)}>
                      Open {region.label} in Brain <span>↗</span>
                    </button>
                  ))}
                </div>
              </>
            ) : (
              <div className="networkSelectionCopy selectionEmpty">
                <span className="detailEyebrow">INTERACTIVE CONNECTOME</span>
                <p>Select any node to inspect its anatomy or a link to inspect the modeled relationship. Associated regions are highlighted for {selectedSystem.name.toLowerCase()}.</p>
              </div>
            )}
          </section>

          <section className="systemOverview">
            <div className="overviewColumn">
              <div className="sectionLabel">OVERVIEW</div>
              <p>{selectedSystem.overview}</p>
              <div className="cautionNote"><span>i</span> These systems are useful functional models, not isolated modules or one-region explanations of behavior.</div>
            </div>
            <div className="overviewColumn">
              <div className="sectionLabel">CORE FUNCTIONS</div>
              <ul className="systemFunctionList">{selectedSystem.functions.map((item) => <li key={item}>{item}</li>)}</ul>
            </div>
          </section>

          <section className="associatedRegions">
            <div className="sectionLabel">ASSOCIATED BRAIN REGIONS</div>
            <div className="linkedSystemRow">
              {selectedSystem.regions.map((id) => {
                const region = findBrainRegion(id)?.region;
                return region ? (
                  <button type="button" className="associatedRegion" key={id} onClick={() => onFocusRegion(id)}>
                    <span>{region.abbr}</span>{region.label}<span className="linkedArrow">↗</span>
                  </button>
                ) : null;
              })}
            </div>
          </section>

          <section className="linkedSystems">
            <div className="sectionLabel">INTERACTS WITH</div>
            <div className="linkedSystemRow">
              {selectedSystem.interactions.map((id) => {
                const system = NEURAL_SYSTEMS.find((item) => item.id === id);
                return system ? <button type="button" className="linkedSystem" key={id} onClick={() => onSelectSystem(id)}><span style={{ background: system.color }} />{system.name}<span className="linkedArrow">↗</span></button> : null;
              })}
            </div>
          </section>
          <div className="systemsFooter">NETWORK RELATIONSHIPS ARE SCHEMATIC · NOT A CONNECTIVITY MEASUREMENT</div>
        </main>
      </div>
    </div>
  );
}
